function doLogin(){
  const email = document.getElementById('loginEmail').value.trim().toLowerCase();
  const password = document.getElementById('loginPassword').value;
  const errEl = document.getElementById('loginError');
  if(!email || !password){ errEl.textContent = 'Preencha e-mail e senha.'; return; }
  const user = USERS.find(u => u.email === email);
  if(!user){ errEl.textContent = 'Não encontramos essa conta. Crie uma em "Criar conta".'; return; }
  if(user.password !== password){ errEl.textContent = 'Senha incorreta.'; return; }
  errEl.textContent = ''; enterApp(user);
}
function doSignup(){

function fmt(v){ return 'R$ ' + Number(v||0).toFixed(2).replace('.', ','); }
function parseBR(dateStr){
  if(!dateStr) return null;
  const p = String(dateStr).split('/');
  if(p.length !== 3) return null;
  return new Date(Number(p[2]), Number(p[1])-1, Number(p[0]));
}
function today(){ const d = new Date(); d.setHours(0,0,0,0); return d; }
function isPendente(r){
  const sit = String(r.situacao||'').trim().toUpperCase();
  if(sit === 'PT') return false; // "PT" = Pago Total na planilha: nunca conta como pendente/em aberto
  return !r.pagamento || Number(r.saldo) > 0;
}
function isVencido(r){
  if(!isPendente(r)) return false;
  const v = parseBR(r.vencimento);
  return v && v < today();
}
function diasAtraso(r){
  const v = parseBR(r.vencimento);
  if(!v) return 0;
  const diff = Math.floor((today() - v) / 86400000);
  return diff > 0 ? diff : 0;
}
function statusOf(r){ if(!isPendente(r)) return 'pago'; return isVencido(r) ? 'vencido' : 'pendente'; }
function pendingValue(r){ return isPendente(r) ? (Number(r.saldo) > 0 ? Number(r.saldo) : Number(r.valor)) : 0; }

  document.getElementById('demonstrativoBody').innerHTML = linhas.map(l=>rowHtml(l,false)).join('') + rowHtml(totalRow, true);

  // checagem de consistência: Valor - despesas deveria bater aproximadamente com Valor Líquido
  const calc = totalRow.valor - totalRow.tarifaVenda - totalRow.tarifaEnvio - totalRow.freteComprador - totalRow.descontosBonus;

  }
}

function renderAlerts(){
  const alerts = [];
  DATA.forEach(r => {
    const dias = diasAtraso(r);
    if(isVencido(r) && dias >= 30) alerts.push({ level:'high', text: `<b>${r.cliente}</b> (${r.revenda}) — título ${r.titulo}: vencido há ${dias} dias, ${fmt(pendingValue(r))} em aberto.` });
    else if(isVencido(r)) alerts.push({ level:'mid', text: `<b>${r.cliente}</b> (${r.revenda}) — título ${r.titulo}: vencido há ${dias} dias.` });
    if(String(r.vendaCancelada).toLowerCase().includes('sim') && isPendente(r)) alerts.push({ level:'high', text: `<b>${r.cliente}</b> (${r.revenda}) — título ${r.titulo}: venda marcada como cancelada, mas ainda consta ${fmt(pendingValue(r))} pendente. Confirmar se o CR foi ajustado.` });
  });
  const seen = new Set();
  DATA.forEach(r => { const key = r.revenda+'|'+r.titulo; if(seen.has(key)) alerts.push({ level:'high', text:`Título duplicado: <b>${r.titulo}</b> aparece mais de uma vez em ${r.revenda}.` }); seen.add(key); });

  document.getElementById('alertCount').textContent = alerts.length ? `${alerts.length} identificados` : 'nenhum encontrado';
  document.getElementById('alertsList').innerHTML = alerts.length
    ? alerts.slice(0,12).map(a => `<div class="alert-item"><span class="alert-dot ${a.level}"></span><span>${a.text}</span></div>`).join('')
    : `<div class="alert-empty">Nenhuma inconsistência identificada nos dados atuais.</div>`;
}

function renderTable(){

    if(revendaSel && r.revenda !== revendaSel) return false;

    return true;
  });
  document.getElementById('tableBody').innerHTML = rows.map(r => {
    const st = statusOf(r);
    const badge = st==='vencido' ? `<span class="badge venc">vencido (${diasAtraso(r)}d)</span>` : st==='pendente' ? `<span class="badge pend">pendente</span>` : `<span class="badge pago">recebido</span>`;
    return `<tr>
      <td>${r.revenda}</td><td>${r.cliente}</td><td>${r.titulo}</td><td>${r.situacao}</td>
      <td>${fmt(r.valor)}</td><td>${r.vencimento||'—'}</td><td>${r.pagamento||'—'}</td>
      <td>${st==='vencido' ? diasAtraso(r) : '—'}</td><td>${fmt(r.valorLiquido)}</td><td>${badge}</td>
    </tr>`;
  }).join('');
}

function excelDateToStr(v){
  if(v instanceof Date) return String(v.getDate()).padStart(2,'0')+'/'+String(v.getMonth()+1).padStart(2,'0')+'/'+v.getFullYear();
  if(typeof v === 'number' && v > 20000 && v < 60000){
    const d = new Date(Date.UTC(1899,11,30) + v*86400000);
    return String(d.getUTCDate()).padStart(2,'0')+'/'+String(d.getUTCMonth()+1).padStart(2,'0')+'/'+d.getUTCFullYear();
  }
  return v ? String(v) : '';
}
function findCol(headers, mustHave, mustNotHave){
  mustNotHave = mustNotHave || [];
  return headers.find(h => {
    const l = h.toLowerCase();
    return mustHave.some(w => l.includes(w)) && !mustNotHave.some(w => l.includes(w));
  });
}
function scoreHeaderRow(arr){
  const lower = arr.map(c => String(c).toLowerCase());
  let score = 0;
  ['nome','cliente','título','titulo','valor','vencimento','pagamento','tarifa'].forEach(w => { if(lower.some(c=>c.includes(w))) score++; });
  return score;
}

function handleUpload(event){
  const file = event.target.files[0];
  if(!file) return;
  const reader = new FileReader();
  reader.onload = function(e){
    try{
      const wb = XLSX.read(e.target.result, { type:'array', cellDates:true });
      let newData = [];
      let unmatchedSheets = [];

      wb.SheetNames.forEach(sheetName => {
        const sheet = wb.Sheets[sheetName];
        const rows = XLSX.utils.sheet_to_json(sheet, { header:1, defval:'', blankrows:false, raw:true });
        if(rows.length < 2) return;

        let bestIdx = 0, bestScore = -1;
        for(let i=0; i<Math.min(rows.length, 12); i++){
          const s = scoreHeaderRow(rows[i]);
          if(s > bestScore){ bestScore = s; bestIdx = i; }
        }
        if(bestScore < 3){ unmatchedSheets.push(sheetName); return; }

        const headers = rows[bestIdx].map(h => String(h));
        const nomeKey = findCol(headers, ['nome']);
        const clienteKey = findCol(headers, ['cliente']);
        const tituloKey = findCol(headers, ['título','titulo']);
        const sitKey = findCol(headers, ['sit']);
        const valorKey = findCol(headers, ['valor'], ['líquido','liquido']);
        const saldoKey = findCol(headers, ['saldo']);
        const emissaoKey = findCol(headers, ['emiss']);
        const vencKey = findCol(headers, ['vencimento']);
        const pagKey = findCol(headers, ['pagamento']);
        const tarifaVendaKey = findCol(headers, ['tarifa'], ['envio']);
        const tarifaEnvioKey = findCol(headers, ['envio']);
        const despesaKey = findCol(headers, ['despesa']);
        const freteKey = findCol(headers, ['frete']);
        const descontoKey = findCol(headers, ['desconto']);
        const liquidoKey = findCol(headers, ['líquido','liquido']);
        const canceladaKey = findCol(headers, ['cancelada']);
        const solucaoKey = findCol(headers, ['solução','solucao']);

        if(!nomeKey || !valorKey || !vencKey){ unmatchedSheets.push(sheetName); return; }

        for(let i=bestIdx+1; i<rows.length; i++){
          const row = rows[i];
          const obj = {}; headers.forEach((h,idx) => obj[h] = row[idx]);
          if(!obj[nomeKey]) continue;
          newData.push({
            revenda: sheetName,
            cliente: String(obj[nomeKey]),
            codCliente: clienteKey ? String(obj[clienteKey]||'') : '',
            titulo: tituloKey ? String(obj[tituloKey]||'') : '',
            situacao: sitKey ? String(obj[sitKey]||'') : '',
            valor: Number(obj[valorKey])||0,
            saldo: saldoKey ? (Number(obj[saldoKey])||0) : 0,
            emissao: emissaoKey ? excelDateToStr(obj[emissaoKey]) : '',
            vencimento: vencKey ? excelDateToStr(obj[vencKey]) : '',
            pagamento: pagKey ? excelDateToStr(obj[pagKey]) : '',
            tarifaVenda: tarifaVendaKey ? (Number(obj[tarifaVendaKey])||0) : 0,
            tarifaEnvio: tarifaEnvioKey ? (Number(obj[tarifaEnvioKey])||0) : 0,
            despesaTotal: despesaKey ? (Number(obj[despesaKey])||0) : 0,
            freteComprador: freteKey ? (Number(obj[freteKey])||0) : 0,
            descontosBonus: descontoKey ? (Number(obj[descontoKey])||0) : 0,
            valorLiquido: liquidoKey ? (Number(obj[liquidoKey])||0) : (Number(obj[valorKey])||0),
            vendaCancelada: canceladaKey ? String(obj[canceladaKey]||'') : '',
            solucao: solucaoKey ? String(obj[solucaoKey]||'') : ''
          });
        }
      });

      if(newData.length === 0){
        showImportModal(`
          <h3>Não consegui reconhecer os títulos nesta planilha</h3>
          <p>Nenhuma aba do arquivo "${file.name}" tinha as colunas mínimas esperadas (Nome/Cliente, Valor, Vencimento). Os dados de exemplo foram mantidos.</p>
          ${unmatchedSheets.length ? `<p class="modal-muted">Abas verificadas sem sucesso: ${unmatchedSheets.join(', ')}</p>` : ''}
          <p class="modal-muted">Colunas mínimas esperadas em cada aba (uma aba por revenda):</p>
          <div class="modal-cols">${REQUIRED_COLS.join(' · ')}</div>
          <div class="modal-actions"><button class="btn btn-gold" onclick="closeImportModal()">Entendi</button></div>
        `);
        return;
      }

      DATA = newData;
      document.getElementById('fileMeta').textContent = `${file.name} — importado agora · ${DATA.length} títulos · ${[...new Set(DATA.map(r=>r.revenda))].length} revendas`;
      renderAll();
      let msg = `Planilha "${file.name}" importada: ${DATA.length} títulos em ${[...new Set(DATA.map(r=>r.revenda))].length} revendas identificados. Dashboard e tabela atualizados.`;
      if(unmatchedSheets.length) msg += ` As abas ${unmatchedSheets.join(', ')} não foram reconhecidas e ficaram de fora.`;
      pushMessage('ai', msg);
    }catch(err){
      showImportModal(`<h3>Erro ao ler o arquivo</h3><p>${err.message}</p><div class="modal-actions"><button class="btn btn-gold" onclick="closeImportModal()">Fechar</button></div>`);
    }
  };
  reader.readAsArrayBuffer(file);
}
